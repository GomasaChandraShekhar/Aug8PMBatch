Variable :: A small memory allocated to store data.

var,   let,    const

var :: 

Scope :: Global :: Accessed within the project.
Re-Declaration :: Allowed
Re-Assignment :: Allowed
Hoisting :: Access the Variable before it is declared.

Syntax ::

var varibaleName: type; // Declaration
var varibaleName: type = value; // Declaration and inilization

varibaleName = dsfdsfdsf' // Assignment


let :: 

Scope :: Function or Block Scoped :: Accessed within the area declared.
Re-Declaration :: Not allowed
Re-Assignment :: Allowed
Hoisting :: Not allowed

Syntax ::

let varibaleName: type; // Declaration
let varibaleName: type = value; // Declaration and inilization

varibaleName = dsfdsfdsf // Assignment


const :: Constant :: Not change :: 

Scope :: Function or Block Scoped :: Accessed within the area declared.
Re-Declaration :: Not allowed
Re-Assignment :: Not allowed
Hoisting :: Not allowed

Syntax ::

const varibaleName: type = value; // Declaration and inilization

Logical Operators :: 

1. Logical AND Operator :: && :: 
        If all inputs are true output is true, else output false.

2. Logical OR Operator :: || :: 
        If at least one input is true output is true, 
        if all inputs are false output is false.

3. Logical NOT Operator :: ! :: 
        if input is true output is false.
        if input is false output is true.


1. Logical AND Operator :: && :: 

in1         &&           in2            op
true        &&          true            true
true        &&          false           false
false       &&          true            false
false       &&          false           false


2. Logical OR Operator :: || :: 

in1         ||           in2            op
true        ||          true            true
true        ||          false           true
false       ||          true            true
false       ||          false           false


3. Logical NOT Operator :: ! :: 

in          !       op
true        !       false
false       !       true


Ternary Operator :: Conditional Operator :: ? :: 

Syntax :: 

let result = Condition ? "message1" : "message2"
